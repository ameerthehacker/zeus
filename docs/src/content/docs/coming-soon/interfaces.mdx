## Planned Features

### Interface Declaration

Define contracts that classes must implement:

```zeus
interface Printable {
  print();
}

interface Comparable {
  compareTo(other: Comparable): i32;
}
```

### Implementing Interfaces

Classes can implement one or more interfaces:

```zeus
interface Shape {
  area(): f64;
  perimeter(): f64;
}

class Rectangle implements Shape {
  public width: f64;
  public height: f64;

  constructor(w: f64, h: f64) {
    this.width = w;
    this.height = h;
  }

  public area(): f64 {
    return this.width * this.height;
  }

  public perimeter(): f64 {
    return 2.0 * (this.width + this.height);
  }
}
```

### Multiple Interfaces

A class can implement multiple interfaces:

```zeus
interface Drawable {
  draw();
}

interface Clickable {
  onClick();
}

class Button implements Drawable, Clickable {
  public draw() {
    // Render the button
  }

  public onClick() {
    // Handle click
  }
}
```

### Interface as Type

Use interfaces as variable and parameter types:

```zeus
function calculateTotalArea(shapes: Shape[]): f64 {
  let total: f64 = 0.0;
  let i: i32 = 0;
  
  while (i < shapes.length()) {
    total = total + shapes.get(i).area();
    i = i + 1;
  }
  
  return total;
}
```

## Use Cases

### Dependency Injection

```zeus
interface Logger {
  log(message: string);
}

class ConsoleLogger implements Logger {
  public log(message: string) {
    // Print to console
  }
}

class FileLogger implements Logger {
  public log(message: string) {
    // Write to file
  }
}

class Application {
  private logger: Logger;

  constructor(logger: Logger) {
    this.logger = logger;
  }
}
```

### Plugin Systems

```zeus
interface Plugin {
  getName(): string;
  execute();
}

function runPlugins(plugins: Plugin[]) {
  let i: i32 = 0;
  while (i < plugins.length()) {
    plugins.get(i).execute();
    i = i + 1;
  }
}
```
